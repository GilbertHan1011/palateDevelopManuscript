def plot_and_annotate_single_point(score_eign_center, degree_centrality_all, gene_to_annotate, x_lim=None, y_lim=None):
    """
    Plot a single point from the DataFrame score_eign_center and annotate it on a scatter plot based on degree_centrality_all.

    Parameters:
    - score_eign_center: DataFrame containing the data
    - degree_centrality_all: Name of the column for degree centrality
    - gene_to_annotate: The gene to annotate
    - x_lim: Tuple specifying the limits for the x-axis (optional)
    - y_lim: Tuple specifying the limits for the y-axis (optional)
    """
    # Filter DataFrame based on gene_to_annotate
    single_gene_df = score_eign_center[score_eign_center['gene'] == gene_to_annotate]
    
    # Extract x and y values for the gene
    x_gene = single_gene_df[degree_centrality_all].values[0]
    y_gene = single_gene_df[degree_centrality_all].values[0]
    
    # Scatter plot
    plt.scatter(score_eign_center[degree_centrality_all], score_eign_center[degree_centrality_all], s=0.8)
    
    # Annotate the single point
    plt.annotate(gene_to_annotate, (x_gene, y_gene), textcoords="offset points", xytext=(0, 10), ha='center')
    
    # Add labels and title
    plt.xlabel(degree_centrality_all)
    plt.ylabel(degree_centrality_all)
    plt.title(f'Scatter plot with {gene_to_annotate} annotation')
    
    # Set axis limits if provided
    if x_lim:
        plt.xlim(*x_lim)
    if y_lim:
        plt.ylim(*y_lim)
    
    # Show plot
    plt.show()
